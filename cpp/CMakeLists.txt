cmake_minimum_required(VERSION 3.18)
project(erg_kernel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Intended for a conda env containing rdkit + librdkit-dev + libboost-devel + pybind11
# (conda create -n erg_cpp -c conda-forge python=3.12 rdkit=2024.09.4 librdkit-dev=2024.09.4 libboost-devel pybind11 numpy cmake ninja)
set(CP $ENV{CONDA_PREFIX})
find_package(pybind11 CONFIG REQUIRED)
find_package(OpenMP)

pybind11_add_module(erg_kernel erg_kernel.cpp)
target_include_directories(erg_kernel PRIVATE ${CP}/Library/include ${CP}/Library/include/rdkit)
target_link_directories(erg_kernel PRIVATE ${CP}/Library/lib)
target_link_libraries(erg_kernel PRIVATE RDKitSmilesParse RDKitGraphMol RDKitRDGeneral RDKitDataStructs RDKitRDGeometryLib)
if(OpenMP_CXX_FOUND)
  target_link_libraries(erg_kernel PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(erg_kernel PRIVATE /O2 /GL /EHsc /wd4244 /wd4267 /wd4996 /bigobj)
else()
  target_compile_options(erg_kernel PRIVATE -O3 -march=native)
endif()
